cmake_minimum_required(VERSION 3.24)
project(CyToolsNativeClient LANGUAGES C CXX)
find_package(CyToolsCommunication 0.11 CONFIG REQUIRED)
add_executable(cytools_cpp_client client.cpp)
target_link_libraries(cytools_cpp_client PRIVATE CyTools::Communication)
add_executable(cytools_c_abi_probe probe.c)
target_link_libraries(cytools_c_abi_probe PRIVATE CyTools::Communication)
if(WIN32)
  foreach(target cytools_cpp_client cytools_c_abi_probe)
    add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:CyTools::Communication> $<TARGET_FILE_DIR:${target}>)
  endforeach()
endif()
