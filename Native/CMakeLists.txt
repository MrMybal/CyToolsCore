cmake_minimum_required(VERSION 3.24)
project(CyToolsCommunication VERSION 0.11.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(CYMCPSDK_SOURCE_DIR "" CACHE PATH "CyMCPSDK 0.7.1 source checkout")
if(CYMCPSDK_SOURCE_DIR)
  set(CYMCP_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(CYMCP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(CYMCP_BUILD_CHAT_UI OFF CACHE BOOL "" FORCE)
  set(CYMCP_BUILD_CHAT_APP OFF CACHE BOOL "" FORCE)
  set(CYMCP_BUILD_AI OFF CACHE BOOL "" FORCE)
  set(CYMCP_BUILD_LOCAL OFF CACHE BOOL "" FORCE)
  add_subdirectory("${CYMCPSDK_SOURCE_DIR}" cymcp)
else()
  find_package(CyMCPSDK 0.7.1 CONFIG REQUIRED COMPONENTS Core Platform Http MCP)
endif()
add_library(cytools_communication SHARED src/communication.cpp src/c_api.cpp)
add_library(CyTools::Communication ALIAS cytools_communication)
set_target_properties(cytools_communication PROPERTIES EXPORT_NAME Communication)
target_compile_features(cytools_communication PUBLIC cxx_std_20)
target_include_directories(cytools_communication PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(cytools_communication PUBLIC CyMCP::Core CyMCP::MCP CyMCP::Http)
target_compile_definitions(cytools_communication PRIVATE CYTOOLS_COMMUNICATION_EXPORTS)
if(MSVC)
  target_compile_options(cytools_communication PRIVATE /W4 /utf-8)
  target_compile_definitions(cytools_communication PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
add_executable(cytools-native src/main.cpp)
target_link_libraries(cytools-native PRIVATE CyTools::Communication)
install(TARGETS cytools_communication cytools-native EXPORT CyToolsTargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT CyToolsTargets NAMESPACE CyTools:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/CyToolsCommunication)
configure_package_config_file(cmake/CyToolsCommunicationConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/CyToolsCommunicationConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/CyToolsCommunication)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/CyToolsCommunicationConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/CyToolsCommunicationConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/CyToolsCommunicationConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/CyToolsCommunication)

install(DIRECTORY ../LICENCES/ DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/CyToolsCommunication)
install(FILES ../LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/CyToolsCommunication RENAME CyToolsCore-MIT.txt)
install(FILES ../Docs/NativeCommunication.md DESTINATION ${CMAKE_INSTALL_DATADIR}/CyToolsCommunication)
